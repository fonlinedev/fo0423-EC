    

void map_init(Map&map,bool firstTime)
{
	if(firstTime)
	{
		Log("Map init "+map.GetProtoId());
		map.SetTurnBasedAvailability(true);
		initLockers(map);
	}
}

void s_Dialog(Critter&critter,Scenery&trigger,bool entered,uint8 dir,int dialogId)
{
	if(!critter.IsPlayer()||!entered)
	return;
	
	RunDialog(critter,dialogId,critter.HexX,critter.HexY,false);
}

void t_FinishInvestigation(Critter&critter,Scenery&trigger,bool entered,uint8 dir)
{
	if(!critter.IsPlayer()||!entered)
	return;
	RunDialog(critter,(4023),critter.HexX,critter.HexY,false);
}

void t_Level3Clues(Critter&critter,Scenery&trigger,bool entered,uint8 dir,int clueId)
{
	if(!critter.IsPlayer()||!entered)
	return;
	critter.SayMsg((11),(1),(1000000000+((4023))*100000+(clueId)));
}

void r_AdvanceQuest(Critter&player,Critter@npc){
	GameVar@quest=GetLocalVar((7327),player.Id);
	if((@quest!=null))
	quest=2;
}

void initLockers(Map&map){  
	
	if(map.GetProtoId()==(35)){
		Log("Init lockers at level1");
		
		AddItemToMapItem(map,60,60,(139),(79),6);
		AddItemToMapItem(map,46,42,(139),(40),4);
		AddItemToMapItem(map,46,42,(139),(47),1);
	}else if(map.GetProtoId()==(36)){
		Log("Init lockers at level2");
		
		AddItemToMapItem(map,87,75,(189),(76),1);
		AddItemToMapItem(map,87,75,(189),(80),1);
		AddItemToMapItem(map,87,75,(189),(75),1);
		AddItemToMapItem(map,63,75,(189),(74),1);
		AddItemToMapItem(map,63,75,(189),(127),1);
		AddItemToMapItem(map,51,76,(189),(91),1);
		AddItemToMapItem(map,51,76,(189),(84),1);
		AddItemToMapItem(map,51,76,(189),(73),1);
	}else if(map.GetProtoId()==(37)){
		Log("Init lockers at level3"); 
		
		AddItemToMapItem(map,54,66,(188),(25),2);
		AddItemToMapItem(map,54,66,(188),(51),1);
		
		AddItemToMapItem(map,52,66,(188),(9),1);
		AddItemToMapItem(map,52,66,(188),(29),50);
		
		AddItemToMapItem(map,77,50,(188),(18),1);
		AddItemToMapItem(map,77,50,(188),(35),50);
		AddItemToMapItem(map,77,50,(188),(48),3);
	}
}

void AddItemToMapItem(Map&map,int hexX,int hexY,int pid,int itemPid,uint count)
{
	Item@item=map.GetItem(hexX,hexY,pid);
	
	if(!(@item!=null))
	return;
	
	item.AddItem(itemPid,count,0);
}
