void _InitRespawnContainer(Item&container,bool firstTime)
{ 
	
	CreateTimeEvent((((60*60))*__TimeMultiplier),"e_RespawnLoot",container.Id,true); 
	
	if(firstTime)
	{
		FillContainerWithRandomLoot(container);
	}
} 

uint e_RespawnLoot(uint[]@values)
{
	uint containerId=values[0];
	Item@container=GetItem(containerId); 
	
	if((@container!=null))
	{
		
		ClearContainer(container); 
		
		FillContainerWithRandomLoot(container);
	} 
	
	return(60*60);
} 

void FillContainerWithRandomLoot(Item&container)
{
	
	int roll=Random(1,100);
	
	if(roll<=50)
	{
		container.AddItem((40),Random(1,3),0);
		container.AddItem((29),Random(10,30),0);
	}
	else if(roll<=85)
	{
		container.AddItem((8),1,0);
		container.AddItem((87),1,0);
	}
	else
	{
		container.AddItem((18),1,0);
		container.AddItem((31),20,0);
		container.AddItem((144),1,0);
	}
} 

void ClearContainer(Item&container)
{
	Item@[]items;
	uint count=container.GetItems(0,items);
	for(uint i=0;i<count;i++)
	{
		DeleteItem(items[i]);
	}
}
